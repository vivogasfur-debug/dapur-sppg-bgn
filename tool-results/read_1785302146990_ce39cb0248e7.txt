    79→    79→const calculateAge = (birthDateString: string) => {
    80→    80→  if (!birthDateString || birthDateString === '-') return '-';
    81→    81→  const birthDate = new Date(birthDateString);
    82→    82→  const today = new Date();
    83→    83→
    84→    84→  if (isNaN(birthDate.getTime())) return '-';
    85→    85→
    86→    86→  let years = today.getFullYear() - birthDate.getFullYear();
    87→    87→  let months = today.getMonth() - birthDate.getMonth();
    88→    88→
    89→    89→  if (months < 0 || (months === 0 && today.getDate() < birthDate.getDate())) {
    90→    90→    years--;
    91→    91→    months += 12;
    92→    92→  }
    93→    93→
    94→    94→  if (years === 0) {
    95→    95→    return `${months} Bln`;
    96→    96→  }
    97→    97→
    98→    98→  return months > 0 ? `${years} Thn ${months} Bln` : `${years} Thn`;
    99→    99→};
