    78→const calculateAge = (birthDateString: string) => {
    79→  if (!birthDateString || birthDateString === '-') return '-';
    80→  const birthDate = new Date(birthDateString);
    81→  const today = new Date();
    82→  if (isNaN(birthDate.getTime())) return '-';
    83→  let years = today.getFullYear() - birthDate.getFullYear();
    84→  let months = today.getMonth() - birthDate.getMonth();
    85→  if (months < 0 || (months === 0 && today.getDate() < birthDate.getDate())) {
    86→    years--;
    87→    months += 12;
    88→  }
    89→  if (years === 0) return `${months} Bln`;
    90→  return months > 0 ? `${years} Thn ${months} Bln` : `${years} Thn`;
    91→};
